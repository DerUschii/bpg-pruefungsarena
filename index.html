BEGIN; CREATE OR REPLACE FUNCTION public.join_game(p_pin text,p_nickname text) RETURNS TABLE(player_id uuid,room_id uuid,nickname text) LANGUAGE plpgsql SECURITY DEFINER SET search_path TO '' AS $body$ #variable_conflict use_column DECLARE r public.game_rooms%ROWTYPE; pid uuid; nick text; guest boolean; BEGIN PERFORM public.kanuut_require_access(); SELECT nickname INTO nick FROM public.kanuut_profiles WHERE user_id=auth.uid(); guest := nick IS NULL; IF guest THEN IF NOT EXISTS (SELECT 1 FROM auth.users u WHERE u.id=auth.uid() AND u.is_anonymous IS TRUE) THEN RAISE EXCEPTION 'Bitte zuerst mit deinem Spieleraccount anmelden'; END IF; nick := btrim(p_nickname); IF nick IS NULL OR char_length(nick) NOT BETWEEN 2 AND 24 OR nick !~ '^[[:alnum:]ÄÖÜäöüß][[:alnum:]ÄÖÜäöüß _.-]*$' THEN RAISE EXCEPTION 'Gast-Nickname muss 2–24 Zeichen haben (Buchstaben, Zahlen, Leerzeichen, _, . oder -)'; END IF; IF EXISTS (SELECT 1 FROM public.kanuut_profiles pr WHERE lower(btrim(pr.nickname))=lower(nick)) THEN RAISE EXCEPTION 'Dieser Nickname gehört einem registrierten Account. Bitte anderen wählen'; END IF; END IF; SELECT * INTO r FROM public.game_rooms WHERE pin=p_pin; IF NOT FOUND THEN RAISE EXCEPTION 'Spielcode ungültig'; END IF; SELECT id INTO pid FROM public.players WHERE room_id=r.id AND user_id=auth.uid(); IF pid IS NOT NULL THEN -- Beim Wiedereinstieg denselben Namen beibehalten; Accounts behalten Profilnamen. IF NOT guest THEN UPDATE public.players SET nickname=nick WHERE id=pid AND nickname IS DISTINCT FROM nick; END IF; RETURN QUERY SELECT p.id,r.id,p.nickname::text FROM public.players p WHERE p.id=pid; RETURN; END IF; IF r.status <> 'waiting' THEN RAISE EXCEPTION 'Spiel bereits gestartet. Nur bestehende Spieler können wieder beitreten'; END IF; IF EXISTS (SELECT 1 FROM public.players p WHERE p.room_id=r.id AND lower(btrim(p.nickname))=lower(nick)) THEN RAISE EXCEPTION 'Nickname in diesem Raum bereits vergeben'; END IF; INSERT INTO public.players(room_id,user_id,nickname) VALUES(r.id,auth.uid(),nick) ON CONFLICT (room_id,user_id) DO UPDATE SET nickname=excluded.nickname RETURNING id INTO pid; RETURN QUERY SELECT pid,r.id,nick; END $body$; COMMIT;

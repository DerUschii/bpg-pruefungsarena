-- BPG Prüfungsarena: öffentliche Raumerstellung mit anonymem Supabase Auth. -- Zuerst in Supabase Authentication > Providers > Anonymous Sign-ins aktivieren. -- Alte Hostfunktionen bleiben bestehen, werden aber von der neuen Seite nicht mehr verwendet. -- Jeder Nutzer darf nur eigene Räume starten; Fragen/Lösungen bleiben durch RPC geschützt. BEGIN; CREATE OR REPLACE FUNCTION public.arena_list_quizzes() RETURNS TABLE(quiz_id uuid, quiz_title text, question_count bigint) LANGUAGE sql SECURITY DEFINER SET search_path = '' AS $$ SELECT q.id, q.title::text, count(qq.id)::bigint FROM public.quizzes q LEFT JOIN public.questions qq ON qq.quiz_id=q.id WHERE auth.uid() IS NOT NULL GROUP BY q.id,q.title ORDER BY q.title; $$; REVOKE ALL ON FUNCTION public.arena_list_quizzes() FROM PUBLIC,anon; GRANT EXECUTE ON FUNCTION public.arena_list_quizzes() TO authenticated; CREATE OR REPLACE FUNCTION public.arena_create_room(p_quiz_id uuid) RETURNS TABLE(room_id uuid, game_pin text) LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$ DECLARE v_pin text; v_room uuid; v_attempt int; BEGIN IF auth.uid() IS NULL THEN RAISE EXCEPTION 'Anmeldung erforderlich'; END IF; IF NOT EXISTS (SELECT 1 FROM public.questions WHERE quiz_id=p_quiz_id) THEN RAISE EXCEPTION 'Quiz nicht gefunden oder leer'; END IF; FOR v_attempt IN 1..30 LOOP v_pin := (floor(random()*900000)+100000)::int::text; BEGIN INSERT INTO public.game_rooms (quiz_id,pin,host_user_id,status) VALUES(p_quiz_id,v_pin,auth.uid(),'waiting') RETURNING id INTO v_room; room_id:=v_room; game_pin:=v_pin; RETURN NEXT; RETURN; EXCEPTION WHEN unique_violation THEN NULL; END; END LOOP; RAISE EXCEPTION 'Kein freier Spielcode gefunden. Bitte erneut versuchen.'; END; $$; REVOKE ALL ON FUNCTION public.arena_create_room(uuid) FROM PUBLIC,anon; GRANT EXECUTE ON FUNCTION public.arena_create_room(uuid) TO authenticated; CREATE OR REPLACE FUNCTION public.arena_start_game(p_room_id uuid) RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$ DECLARE v_updated integer; BEGIN IF auth.uid() IS NULL THEN RAISE EXCEPTION 'Anmeldung erforderlich'; END IF; UPDATE public.game_rooms SET status='playing',question_started_at=clock_timestamp() WHERE id=p_room_id AND host_user_id=auth.uid() AND status='waiting'; GET DIAGNOSTICS v_updated=ROW_COUNT; RETURN v_updated=1; END; $$; REVOKE ALL ON FUNCTION public.arena_start_game(uuid) FROM PUBLIC,anon; GRANT EXECUTE ON FUNCTION public.arena_start_game(uuid) TO authenticated; CREATE OR REPLACE FUNCTION public.arena_room_players(p_pin text) RETURNS TABLE(player_name text,player_score integer) LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$ DECLARE v_room public.game_rooms%ROWTYPE; BEGIN IF auth.uid() IS NULL THEN RAISE EXCEPTION 'Anmeldung erforderlich'; END IF; SELECT * INTO v_room FROM public.game_rooms WHERE pin=p_pin; IF NOT FOUND THEN RAISE EXCEPTION 'Spielraum nicht gefunden'; END IF; IF v_room.host_user_id IS DISTINCT FROM auth.uid() AND NOT EXISTS (SELECT 1 FROM public.players p WHERE p.room_id=v_room.id AND p.user_id=auth.uid()) THEN RAISE EXCEPTION 'Kein Zugriff auf diesen Spielraum'; END IF; RETURN QUERY SELECT p.nickname::text,p.score::integer FROM public.players p WHERE p.room_id=v_room.id ORDER BY p.score DESC,p.joined_at ASC; END; $$; REVOKE ALL ON FUNCTION public.arena_room_players(text) FROM PUBLIC,anon; GRANT EXECUTE ON FUNCTION public.arena_room_players(text) TO authenticated; -- Live-Status aus der zuletzt funktionierenden Timer-Version, jetzt mit Raum-Eigentümer als Host. create or replace function public.live_game_state(p_pin text) returns jsonb language plpgsql security definer set search_path = '' as $$ declare v_room public.game_rooms%rowtype; v_player uuid; v_host boolean := false; v_count integer; v_elapsed double precision; v_index integer; v_phase text; v_remaining integer; v_q public.questions%rowtype; v_answer integer; v_board jsonb; v_out jsonb; begin if auth.uid() is null then raise exception 'Anmeldung erforderlich'; end if; select * into v_room from public.game_rooms where pin = p_pin; if not found then raise exception 'Spielraum nicht gefunden'; end if; select id into v_player from public.players where room_id=v_room.id and user_id=auth.uid() limit 1; v_host := v_room.host_user_id=auth.uid(); if v_player is null and not v_host then raise exception 'Kein Zugriff auf diesen Spielraum'; end if; select count(*) into v_count from public.questions where quiz_id=v_room.quiz_id; v_out := jsonb_build_object('status',v_room.status,'pin',v_room.pin,'question_count',v_count, 'server_time',clock_timestamp(),'is_host',v_host); if v_room.status='waiting' or v_room.question_started_at is null then return v_out || jsonb_build_object('phase','waiting','question_number',0,'remaining_seconds',0); end if; if v_count=0 then return v_out || jsonb_build_object('phase','finished','question_number',0,'remaining_seconds',0); end if; -- 25 s Antworten + 3 s Lösung + 3 s Rangliste v_elapsed := greatest(0,extract(epoch from (clock_timestamp()-v_room.question_started_at))); v_index := floor(v_elapsed/31)::integer; if v_index>=v_count then v_phase:='finished';v_remaining:=0; else if mod(v_elapsed::numeric,31)<25 then v_phase:='question';v_remaining:=ceil(25-mod(v_elapsed::numeric,31)); elsif mod(v_elapsed::numeric,31)<28 then v_phase:='reveal';v_remaining:=ceil(28-mod(v_elapsed::numeric,31)); else v_phase:='leaderboard';v_remaining:=ceil(31-mod(v_elapsed::numeric,31)); end if; end if; if v_phase='finished' or v_phase='leaderboard' then select coalesce(jsonb_agg(jsonb_build_object('name',p.nickname,'score',p.score) order by p.score desc,p.joined_at asc), '[]'::jsonb) into v_board from public.players p where p.room_id=v_room.id; v_out:=v_out || jsonb_build_object('leaderboard',v_board); end if; if v_phase='finished' then return v_out || jsonb_build_object('phase','finished','question_number',v_count,'remaining_seconds',0); end if; select * into v_q from public.questions q where q.quiz_id=v_room.quiz_id order by q.position nulls last,q.id offset v_index limit 1; if v_player is not null then select a.selected_index into v_answer from public.answers a where a.player_id=v_player and a.question_id=v_q.id; end if; v_out:=v_out || jsonb_build_object('phase',v_phase,'question_number',v_index+1, 'remaining_seconds',v_remaining,'question',v_q.question,'options',v_q.options, 'my_answer',v_answer); if v_phase='reveal' or v_phase='leaderboard' then v_out:=v_out || jsonb_build_object('correct_index',v_q.correct_index,'explanation',v_q.explanation); end if; return v_out; end; $$; revoke all on function public.live_game_state(text) from public,anon; grant execute on function public.live_game_state(text) to authenticated; create or replace function public.live_submit_answer(p_pin text,p_selected_index integer) returns jsonb language plpgsql security definer set search_path = '' as $$ declare v_room public.game_rooms%rowtype; v_player public.players%rowtype; v_q public.questions%rowtype; v_count integer; v_elapsed double precision; v_index integer; v_phase_elapsed double precision; v_points integer; begin if auth.uid() is null then raise exception 'Anmeldung erforderlich'; end if; if p_selected_index not between 0 and 3 then raise exception 'Ungültige Antwort'; end if; select * into v_room from public.game_rooms where pin=p_pin; if not found or v_room.status<>'playing' or v_room.question_started_at is null then raise exception 'Spiel ist nicht gestartet'; end if; -- Sperrt diesen Spieler, damit parallele Abgaben nicht doppelt punkten. select * into v_player from public.players where room_id=v_room.id and user_id=auth.uid() for update; if not found then raise exception 'Du bist kein Spieler in diesem Raum'; end if; v_elapsed:=extract(epoch from (clock_timestamp()-v_room.question_started_at)); if v_elapsed<0 then raise exception 'Frage noch nicht gestartet'; end if; v_index:=floor(v_elapsed/31)::integer; v_phase_elapsed:=mod(v_elapsed::numeric,31); select count(*) into v_count from public.questions where quiz_id=v_room.quiz_id; if v_index>=v_count or v_phase_elapsed>=25 then raise exception 'Antwortzeit abgelaufen'; end if; select * into v_q from public.questions q where q.quiz_id=v_room.quiz_id order by q.position nulls last,q.id offset v_index limit 1; if exists(select 1 from public.answers where player_id=v_player.id and question_id=v_q.id) then raise exception 'Frage bereits beantwortet'; end if; -- 500 Basispunkte + bis zu 500 Zeitbonus, nur für richtige Antworten. v_points:=case when p_selected_index=v_q.correct_index then 500+floor(500*(25-v_phase_elapsed)/25)::integer else 0 end; insert into public.answers(room_id,player_id,question_id,selected_index) values(v_room.id,v_player.id,v_q.id,p_selected_index); update public.players set score=score+v_points where id=v_player.id; -- Absichtlich weder korrekte Lösung noch Punktzahl vor Ablauf preisgeben. return jsonb_build_object('accepted',true); end; $$; revoke all on function public.live_submit_answer(text,integer) from public,anon; grant execute on function public.live_submit_answer(text,integer) to authenticated; -- Zusätzliche Härtung: direkte Lese-/Schreibrechte bleiben gesperrt. revoke all on public.questions,public.answers,public.players,public.game_rooms from anon,authenticated; COMMIT;
